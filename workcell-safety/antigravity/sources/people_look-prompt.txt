def connect_texture(material, channel, path, vector=None, uv_name=None):
    nt = material.node_tree
    shader = nt.nodes.get('Principled BSDF')
    tex = nt.nodes.new('ShaderNodeTexImage')
    tex.label=channel
    tex.image = bpy.data.images.load(str(path), check_existing=True)
    tex.image.colorspace_settings.name='sRGB' if channel=='Base Color' else 'Non-Color'
    if vector: nt.links.new(vector,tex.inputs['Vector'])
    if channel=='Normal':
        nm=nt.nodes.new('ShaderNodeNormalMap')
        nm.inputs['Strength'].default_value=.85
        if uv_name: nm.uv_map=uv_name
        nt.links.new(tex.outputs['Color'],nm.inputs['Color'])
        nt.links.new(nm.outputs['Normal'],shader.inputs['Normal'])
    elif channel=='ORM':
        split=nt.nodes.new('ShaderNodeSeparateColor')
        nt.links.new(tex.outputs['Color'],split.inputs['Color'])
        for i,d in ((1,'Roughness'),(2,'Metallic')):
            if not shader.inputs[d].is_linked: nt.links.new(split.outputs[i],shader.inputs[d])
    else:
        nt.links.new(tex.outputs['Color'],shader.inputs[channel])

def pbr(key,tile_m=1.5,projection='XY'):
    library='Genie Sim' if key in ARCH else 'Poly Haven'
    name=f'{library} | {key} | {tile_m:.4f} m | {projection}'
    if name in bpy.data.materials:return bpy.data.materials[name]
    if tile_m<=0 or projection not in ('XY','XZ','YZ'):
        raise ValueError((tile_m,projection))
    m=new_material(name)
    nt=m.node_tree
    uv=nt.nodes.new('ShaderNodeUVMap')
    uv.uv_map=SURFACE_UV
    m['finish_projection']=projection
    m['repeat_width_m']=tile_m
    aspect=ARCH[key]['image_aspect_h_over_w'] if key in ARCH else 1.
    m['repeat_height_m']=tile_m*aspect
    for channel,path in finish_maps(key).items():
        connect_texture(m,channel,path,uv.outputs['UV'],SURFACE_UV)
    if key in ARCH:
        m['credit']='AgiBot Genie Sim Assets | CC BY-NC-SA 4.0'
        m['original_material']=ARCH[key]['usd_material']
        m['original_albedo']=ARCH[key]['channels']['Base Color']
    else:
        m['credit']='Poly Haven, CC0 1.0: https://polyhaven.com/a/'+key
    return m

def align_planar_uvs():
    """Prepare a tangent frame and map each static finish at its true metre scale.

    World coordinates orient neighbouring panels consistently. End-grain and slab
    edges use their own face plane so grazing light still shows their surface.
    """
    bpy.context.view_layer.update()
    axes={'XY':(0,1),'XZ':(0,2),'YZ':(1,2)}
    dominant=('YZ','XZ','XY')
    for obj in bpy.context.scene.objects:
        if obj.type!='MESH':
            continue
        mesh=obj.data
        if not any(m and 'finish_projection' in m for m in mesh.materials):
            continue
        layer=mesh.uv_layers.get(SURFACE_UV) or mesh.uv_layers.new(name=SURFACE_UV)
        values=np.zeros((len(mesh.loops),2),dtype=np.float32)
        coords=np.empty((len(mesh.vertices),3),dtype=np.float32)
        mesh.vertices.foreach_get('co',coords.ravel())
        matrix=np.array(obj.matrix_world)
        coords=coords @ matrix[:3,:3].T+matrix[:3,3]
        inv_transpose=obj.matrix_world.to_3x3().inverted().transposed()
        for face in mesh.polygons:
            mat=mesh.materials[face.material_index]
            if not mat or 'finish_projection' not in mat:
                continue
            projection=mat['finish_projection']
            normal=np.array((inv_transpose @ face.normal).normalized())
            plane_axis=({0,1,2}-set(axes[projection])).pop()
            if abs(normal[plane_axis])<.66:
                projection=dominant[int(np.argmax(np.abs(normal)))]
            mesh_points=coords[np.array(face.vertices)]
            values[np.array(face.loop_indices)]=mesh_points[:,axes[projection]]/[mat['repeat_width_m'],mat['repeat_height_m']]
        layer.uv.foreach_set('vector',values.ravel())
        mesh.update()
